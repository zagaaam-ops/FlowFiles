/// Represents a single file or directory within FlowFiles.
///
/// This entity is immutable and contains only the information
/// required by the application's domain layer.
///
/// UI-specific or platform-specific logic must not be added here.
class FileEntity {
  /// Creates a new [FileEntity].
  const FileEntity({
    required this.path,
    required this.name,
    required this.isDirectory,
    required this.size,
    required this.lastModified,
    required this.extension,
  });

  /// Full absolute path.
  final String path;

  /// File or folder name.
  final String name;

  /// True when this entity represents a directory.
  final bool isDirectory;

  /// Size in bytes.
  ///
  /// Directories typically return 0.
  final int size;

  /// Last modification date.
  final DateTime lastModified;

  /// File extension without the leading dot.
  ///
  /// Example:
  /// pdf
  /// jpg
  /// dwg
  final String extension;

  /// Returns true if the entity represents a regular file.
  bool get isFile => !isDirectory;

  @override
  String toString() {
    return 'FileEntity('
        'name: $name, '
        'path: $path, '
        'directory: $isDirectory'
        ')';
  }

  @override
  bool operator ==(Object other) =>
      identical(this, other) ||
      other is FileEntity &&
          runtimeType == other.runtimeType &&
          path == other.path;

  @override
  int get hashCode => path.hashCode;
}
