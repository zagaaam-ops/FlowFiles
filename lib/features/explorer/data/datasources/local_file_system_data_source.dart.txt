import 'dart:io';

/// Reads folders and files from the local file system.
///
/// This class is responsible only for interacting with the operating
/// system. It does not contain business logic.
///
/// The repository layer converts these raw file system objects into
/// domain entities.
class LocalFileSystemDataSource {
  /// Returns a [Directory] for the given path.
  Directory getDirectory(String path) {
    return Directory(path);
  }

  /// Checks whether a directory exists.
  Future<bool> directoryExists(String path) async {
    return Directory(path).exists();
  }

  /// Lists all files and folders inside a directory.
  ///
  /// Hidden files are included by default.
  Future<List<FileSystemEntity>> listDirectory(String path) async {
    final directory = Directory(path);

    if (!await directory.exists()) {
      throw FileSystemException(
        'Directory does not exist.',
        path,
      );
    }

    return directory.list(followLinks: false).toList();
  }
}
